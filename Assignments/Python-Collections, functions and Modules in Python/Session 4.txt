Q-3 : <p>Write a function `reschedule(routes, index, new_time)` in 'train_timetable.py' that returns a new `routes` tuple in which only the route at that index has the new departure time. If the index is not valid, print 'Invalid route number' and return `routes` unchanged. Print the old and new tuples to show the change. Constraint: Do not assign to a tuple; convert with `list()` and back with `tuple()`.</p>
Ans:

routes = (
    ("Sabarmati Express", "Ahmedabad", "Varanasi", "06:30", 1450, True),
    ("Girnar Express", "Ahmedabad", "Mumbai", "08:15", 490, True),
    ("Aravali Express", "Ahmedabad", "Jaipur", "10:45", 650, False),
    ("Ganga Express", "Surat", "Delhi", "14:20", 1150, True),
    ("Deccan Express", "Mumbai", "Pune", "17:00", 192, False),
    ("Narmada Express", "Ahmedabad", "Indore", "21:30", 550, True)
)


def reschedule(routes, index, new_time):

    if index < 0 or index >= len(routes):
        print("Invalid route number")
        return routes

    route_list = list(routes)

    selected_route = list(route_list[index])
    selected_route[3] = new_time

    route_list[index] = tuple(selected_route)

    return tuple(route_list)


print("Old Routes:")
print(routes)

new_routes = reschedule(routes, 1, "09:30")

print("\nNew Routes:")
print(new_routes)

def reschedule(routes, index, new_time):

    if index < 0 or index >= len(routes):
        print("Invalid route number")
        return routes

    route_list = list(routes)

    selected_route = list(route_list[index])
    selected_route[3] = new_time

    route_list[index] = tuple(selected_route)

def fare(route):

    distance = route[4]
    ac_available = route[5]

    if ac_available == True:
        amount = distance * 2.10
    else:
        amount = distance * 1.20

    return round(amount, 2)
