Q-1 : <p>Create 'society_register.py' with a function `add_flat(register, wing, flat, owner, due)` that stores {'owner': owner, 'due': due} at `register[wing][flat]`, creating the wing's inner dictionary only if the wing is new. Use a while loop that asks for these details, ends when the wing is entered as 'done', and builds `register` for a fictional society. Test with 6 flats in at least 2 wings. Hint: `if wing not in register: register[wing] = {}`.</p>
Ans:

def add_flat(register, wing, flat, owner, due):

    if wing not in register:
        register[wing] = {}

    register[wing][flat] = {
        "owner": owner,
        "due": due
    }


register = {}

while True:

    wing = input("Enter wing name (or 'done' to finish): ")

    if wing.lower() == "done":
        break

    flat = input("Enter flat number: ")
    owner = input("Enter owner name: ")
    due = int(input("Enter pending due amount: "))

    add_flat(register, wing, flat, owner, due)

    print("Flat added successfully!\n")


print("\n===== Society Register =====")

for wing, flats in register.items():
    print("\nWing:", wing)

    for flat, details in flats.items():
        print("Flat:", flat)
        print("Owner:", details["owner"])
        print("Due Amount: ₹", details["due"])

def record_payment(register, wing, flat, month, amount):

    if wing in register and flat in register[wing]:

            register[wing][flat]["payments"] = {}

Q-3 : <p>Finish 'society_register.py' with a report that loops through `register.items()` and the flats inside each wing. For every wing, print the flats, the total dues, the total paid and the outstanding amount, followed by the outstanding amount for the whole society. Run it on your 6 flats and 5 payments and paste the output as a comment at the end of the file. Hint: A flat with no payments yet has no 'payments' key, so check `'payments' in details` first.</p>
Ans:

def add_flat(register, wing, flat, owner, due):

    if wing not in register:
        register[wing] = {}

    register[wing][flat] = {
        "owner": owner,
        "due": due
    }


def record_payment(register, wing, flat, month, amount):

    if wing in register and flat in register[wing]:

        if "payments" not in register[wing][flat]:
            register[wing][flat]["payments"] = {}

        register[wing][flat]["payments"][month] = amount
        print("Payment recorded successfully!")

    else:
        print("Flat not found")


register = {}

add_flat(register, "A", "A101", "Rahul", 2000)
add_flat(register, "A", "A102", "Priya", 0)
add_flat(register, "A", "A103", "Amit", 1500)

add_flat(register, "B", "B101", "Neha", 3000)
add_flat(register, "B", "B102", "Meera", 500)
add_flat(register, "B", "B103", "Rohan", 1000)

record_payment(register, "A", "A101", "January", 2000)
record_payment(register, "A", "A101", "February", 1500)
record_payment(register, "A", "A102", "January", 1000)
record_payment(register, "B", "B101", "January", 3000)
record_payment(register, "C", "C101", "January", 1200)

society_outstanding = 0

print("\n========== SOCIETY REPORT ==========")

for wing, flats in register.items():

    total_dues = 0
    total_paid = 0

    print("\nWing:", wing)

    for flat, details in flats.items():

        due = details["due"]
        paid = 0

        if "payments" in details:
            paid = sum(details["payments"].values())

        outstanding = due - paid

        total_dues += due
        total_paid += paid

        print(
            flat,
            "| Owner:", details["owner"],
            "| Due: ₹", due,
            "| Paid: ₹", paid,
            "| Outstanding: ₹", outstanding
        )

    wing_outstanding = total_dues - total_paid
    society_outstanding += wing_outstanding

    print("Total Dues: ₹", total_dues)
    print("Total Paid: ₹", total_paid)
    print("Outstanding: ₹", wing_outstanding)

print("\nWhole Society Outstanding: ₹", society_outstanding)
